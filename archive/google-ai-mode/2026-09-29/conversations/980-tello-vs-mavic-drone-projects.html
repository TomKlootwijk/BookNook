        for (size_t i = msg->ranges.size() / 2 - 2; i < msg->ranges.size() / 2 + 2; ++i) {
            if (!std::isnan(msg->ranges[i])) {
                sum += msg->ranges[i];
                count++;
            }

            current_depth_metric_ = sum / count;
        }
